## 📝 PowerShell 安全清理脚本

```powershell
# ============================================
# 安全清理脚本 v1.0
# 只删除已验证无害的文件
# ============================================

Write-Host "🔍 开始安全检查..." -ForegroundColor Cyan

# 1. 删除根目录临时输出文件
Write-Host "`n📁 删除临时输出文件..." -ForegroundColor Yellow
$tempFiles = @(
    "check_result.txt",
    "geometry_ggb_coord.txt",
    "kg_check_output.txt",
    "kg_gap_report.json",
    "kg_gap_report.txt",
    "resource_recommendation_results.json",
    "vector_db_ggb_coord.txt",
    "vector_search_result.txt",
    "测试结果",
    "check_kp_result.json"
)

foreach ($file in $tempFiles) {
    if (Test-Path $file) {
        Remove-Item $file -Force
        Write-Host "  ✅ 已删除: $file" -ForegroundColor Green
    }
}

# 2. 创建 archive 目录
Write-Host "`n📁 创建归档目录..." -ForegroundColor Yellow
if (-not (Test-Path "scripts\archive")) {
    New-Item -ItemType Directory -Path "scripts\archive" -Force | Out-Null
    Write-Host "  ✅ 已创建: scripts\archive" -ForegroundColor Green
}

# 3. 移动根目录调试脚本到 archive
Write-Host "`n📁 移动根目录调试脚本..." -ForegroundColor Yellow
$scriptPatterns = @("check_*.py", "test_*.py", "find_*.py", "diagnose_*.py", "quick_*.py", "read_*.py", "rebuild_*.py", "verify_*.py")

foreach ($pattern in $scriptPatterns) {
    Get-ChildItem -Path "." -Filter $pattern -ErrorAction SilentlyContinue | ForEach-Object {
        Move-Item $_.FullName -Destination "scripts\archive\" -Force
        Write-Host "  ✅ 已移动: $($_.Name)" -ForegroundColor Green
    }
}

# 移动其他一次性脚本
$otherScripts = @(
    "add_edges.py", "analyze_exercises.py", "analyze_exercises_fast.py", "analyze_log.py",
    "build_database.py", "compare_two_coursewares.py", "continue_analysis.py",
    "delete_empty_analysis.py", "delete_wrong_complex_review.py",
    "fix_lesson_case_titles.py", "invest.py", "load_exercise_analysis.py",
    "merge.py", "ppt_to_chroma.py", "query_chroma.py", "query_chroma2.py",
    "remove_duplicates.py", "search_complex_review_in_db.py",
    "validate_exercise_analysis.py", "verify_graph.py", "verify_resource_index.py"
)

foreach ($script in $otherScripts) {
    if (Test-Path $script) {
        Move-Item $script -Destination "scripts\archive\" -Force
        Write-Host "  ✅ 已移动: $script" -ForegroundColor Green
    }
}

# 4. 创建 backend archive 目录
Write-Host "`n📁 创建 backend 归档目录..." -ForegroundColor Yellow
if (-not (Test-Path "backend\scripts\archive")) {
    New-Item -ItemType Directory -Path "backend\scripts\archive" -Force | Out-Null
    Write-Host "  ✅ 已创建: backend\scripts\archive" -ForegroundColor Green
}

# 5. 移动 backend 调试脚本
Write-Host "`n📁 移动 backend 调试脚本..." -ForegroundColor Yellow
Set-Location backend

$backendPatterns = @("check_*.py", "test_*.py", "debug_*.py", "fix_*.py")

foreach ($pattern in $backendPatterns) {
    Get-ChildItem -Path "." -Filter $pattern -ErrorAction SilentlyContinue | ForEach-Object {
        Move-Item $_.FullName -Destination "scripts\archive\" -Force
        Write-Host "  ✅ 已移动: $($_.Name)" -ForegroundColor Green
    }
}

# 移动 backend 其他一次性脚本
$backendScripts = @(
    "add_debug.py", "add_debug2.py", "add_merge_debug.py", "build_index.py",
    "comprehensive_test.py", "count_placeholders.py", "create_correct_cache.py",
    "create_html.py", "create_sanitized_cache.py", "detect_encoding.py",
    "find_associated.md", "find_target_csv.py", "find_target_in_db.py",
    "find_target_in_index.py", "find_target_row.py", "force_refresh_cache.py",
    "integrate_knowledge_graph.py", "list_collections.py", "list_lesson_plans.py",
    "modify_file.py", "optimize_retrieval.py", "read_prob_csv.py",
    "rebuild_and_verify.py", "rebuild_db.py", "rebuild_vector_db.py",
    "search_monotonicity.py", "simplify_separate.py",
    "verify_all_courseware.py", "verify_cloud_lesson_plan_index.py", "verify_improvement.py"
)

foreach ($script in $backendScripts) {
    if (Test-Path $script) {
        Move-Item $script -Destination "scripts\archive\" -Force
        Write-Host "  ✅ 已移动: $script" -ForegroundColor Green
    }
}

Set-Location ..

# 6. 删除空目录
Write-Host "`n📁 删除空目录..." -ForegroundColor Yellow
$emptyDirs = @("exports", "logs", "test_chroma_db")

foreach ($dir in $emptyDirs) {
    if (Test-Path $dir) {
        $items = Get-ChildItem $dir -Recurse
        if ($items.Count -eq 0) {
            Remove-Item $dir -Recurse -Force
            Write-Host "  ✅ 已删除: $dir" -ForegroundColor Green
        }
    }
}

Write-Host "`n✅ 清理完成！" -ForegroundColor Green
Write-Host "`n📊 统计信息：" -ForegroundColor Cyan
Write-Host "  - 临时文件已删除" -ForegroundColor White
Write-Host "  - 调试脚本已移动到 scripts\archive\ 和 backend\scripts\archive\" -ForegroundColor White
Write-Host "  - 空目录已删除" -ForegroundColor White
Write-Host "`n⚠️  重要提示：" -ForegroundColor Yellow
Write-Host "  - 所有脚本都已移动到 archive 目录，没有永久删除" -ForegroundColor White
Write-Host "  - 如果需要恢复，可以从 archive 目录移回" -ForegroundColor White
Write-Host "  - 建议先测试项目功能是否正常，确认无误后再考虑永久删除" -ForegroundColor White
```
